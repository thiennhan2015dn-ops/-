    SESSION.saves=saves
    activeSlot=i; SESSION.activeSlot=i

local function snapshot()
    return {
        speed=SPEED,
        activeSlot=activeSlot,
        autoReturn=autoReturn,
        saves={saves[1],saves[2],saves[3]}
    }
end
local function apply(p)
    if not p then return end
    SPEED=p.speed or SPEED
    activeSlot=p.activeSlot or activeSlot
    autoReturn=p.autoReturn or autoReturn
    saves={p.saves[1],p.saves[2],p.saves[3]}
    SESSION.speed=SPEED
    SESSION.activeSlot=activeSlot
    SESSION.autoReturn=autoReturn
    SESSION.saves=saves
    speedLbl.Text="SPEED: "..SPEED
    knob.Position=UDim2.new((SPEED-8)/32,-7,-0.4,0)
    auto.Text=autoReturn and "AUTO RETURN: ON" or "AUTO RETURN: OFF"
    refreshSlots()
end

local function selectPreset(k)
    currentPreset=k; SESSION.preset=k
    setChip("PRESET SELECTED", Color3.fromRGB(120,120,255))
end

saveP.MouseButton1Click:Connect(function()

end)

    apply(SESSION.presets[currentPreset])
    setChip("PRESET LOADED", Color3.fromRGB(255,170,0))
end)

-- LOGIC
local function smooth(cf)
    if not cf then return end
    local dist=(hrp.Position-cf.Position).Magnitude
    if dist<MIN_DIST then
        returning=false; bar.Size=UDim2.new(1,0,1,0); pct.Text="100%"
        setChip("ARRIVED",Color3.fromRGB(0,200,0)); return
    end
    if activeTween then activeTween:Cancel() end
    startPos=hrp.Position; totalDist=dist
    activeTween=TweenService:Create(hrp,TweenInfo.new(dist/math.max(SPEED,1),Enum.EasingStyle.Linear),{CFrame=cf})
    returning=true; setChip("RETURNING",Color3.fromRGB(255,170,0))
    activeTween:Play()
end
go.MouseButton1Click:Connect(function() smooth(saves[activeSlot]) end)
auto.MouseButton1Click:Connect(function()
    autoReturn=not autoReturn
    SESSION.autoReturn=autoReturn
    auto.Text=autoReturn and "AUTO RETURN: ON" or "AUTO RETURN: OFF"
end)

-- SLIDER
local dragging=false

knob.InputEnded:Connect(function() dragging=false end)

RunService.RenderStepped:Connect(function()
    if dragging then
        local x=math.clamp((UIS:GetMouseLocation().X-sb.AbsolutePosition.X)/sb.AbsoluteSize.X,0,1)
        knob.Position=UDim2.new(x,-7,-0.4,0)
        SPEED=math.floor(8+x*32)
        SESSION.speed=SPEED
        speedLbl.Text="SPEED: "..SPEED
    end
    if returning and saves[activeSlot] and startPos then
        local cur=(hrp.Position-saves[activeSlot].Position).Magnitude
        local done=math.clamp(1-(cur/math.max(totalDist,1)),0,1)
        bar.Size=UDim2.new(done,0,1,0); pct.Text=math.floor(done*100).."%"
    end
    if autoReturn and saves[activeSlot] and not returning then
        if (hrp.Position-saves[activeSlot].Position).Magnitude>AUTO_DIST then
            smooth(saves[activeSlot])
        end
    end
end)

float.MouseButton1Click:Connect(function() main.Visible=not main.Visible end)

refreshSlots()
print("✅ UI PRO + PRESET MANAGER LOADED")
